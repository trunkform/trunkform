func (DevopsPrompt) Register(s *server.MCPServer) {
        logf.Debugf("DevopsPrompt.Register: registering devops prompt", "\x1b[90m")
        p := mcp.NewPrompt(
                "devops",
                mcp.WithPromptDescription("DevOps-oriented review prompt."),
                mcp.WithArgument("task"),
        )

        s.AddPrompt(p, func(ctx context.Context, req mcp.GetPromptRequest) (*mcp.GetPromptResult, error) {

                logf.Debugf("devops prompt: returning result for task '" + task + "'", "\x1b[90m")
                return mcp.NewGetPromptResult(
                        "DevOps review prompt",
                        []mcp.PromptMessage{
                                mcp.NewPromptMessage(mcp.RoleUser, mcp.NewTextContent(task)),
                        },
                ), nil
        })
        logf.Debugf("DevopsPrompt.Register: devops prompt registered", "\x1b[90m")
}

import (

type LogLevel int

const (
        FATAL LogLevel = 1
        ERROR LogLevel = 2
        WARN  LogLevel = 3
        INFO  LogLevel = 4
        DEBUG LogLevel = 5
        TRACE LogLevel = 6
)

type Logger struct {
        level  LogLevel
        writer io.Writer
}

var logf = NewLogger(os.Getenv("TRUNKFORM_LOGLEVEL"), os.Stderr)

func NewLogger(levelStr string, w io.Writer) *Logger {
        levelMap := map[string]LogLevel{
                "1": FATAL, "2": ERROR, "3": WARN,
                "4": INFO, "5": DEBUG, "6": TRACE,
                "FATAL": FATAL, "ERROR": ERROR, "WARN": WARN,
                "INFO": INFO, "DEBUG": DEBUG, "TRACE": TRACE,
        }
        level, ok := levelMap[levelStr]
        if !ok {
                level = INFO
        }
        return &Logger{level: level, writer: w}
}

func (l *Logger) log(level LogLevel, prefix, message string, colorCode ...string) {
        if level > l.level {
                return
        }
        timestamp := time.Now().Format("2006/01/02 15:04:05")
        color := ""
        if len(colorCode) > 0 {
                color = colorCode[0]
        }
        fmt.Fprintf(l.writer, "%s [%s] %s%s\x1b[0m\n", timestamp, prefix, color, message)
}

func (l *Logger) Fatalf(message string, colorCode ...string)  { l.log(FATAL, "FATAL", message, colorCode...) }
func (l *Logger) Errorf(message string, colorCode ...string)  { l.log(ERROR, "ERROR", message, colorCode...) }
func (l *Logger) Warnf(message string, colorCode ...string)   { l.log(WARN, "WARN", message, colorCode...) }
func (l *Logger) Infof(message string, colorCode ...string)   { l.log(INFO, "INFO", message, colorCode...) }
func (l *Logger) Debugf(message string, colorCode ...string)  { l.log(DEBUG, "DEBUG", message, colorCode...) }
func (l *Logger) Tracef(message string, colorCode ...string)  { l.log(TRACE, "TRACE", message, colorCode...) }

type formatArgs struct {
        Version  string                 `json:"version"`
        Prompt   string                 `json:"prompt"`
        Steering string                 `json:"steering"`
        Related  []string               `json:"related"`
        Rubric   map[string]interface{} `json:"rubric"`
}

func processRubric(args formatArgs) (*mcp.CallToolResult, error) {
        logf.Tracef("processRubric: checking " + fmt.Sprintf("%d", len(args.Rubric)) + " rubric items", "\x1b[90m")
        rubricOrder := []struct {
                key     string
                name    string
                handler func() (string, map[string]any)
        }{
                {"ci-cd-automation-boilerplate", "CI/CD Automation Boilerplate", handleCICDBoilerplate},
                {"perf-test-implemented", "Add Performance Testing", handlePerfTest},
                {"unit-test-implemented", "Add Unit Test Coverage", handleUnitTestCoverage},
                {"continuous-integration-iac", "Add Continuous Integration IAC", handleCIIaC},
                {"continuous-integration-test-double-implemented", "Add Continuous Integration Test Doubles", handleTestDoubles},
                {"continuous-integration-mocks-implemented", "Add Continuous Integration Mock Implementations", handleMocks},
        }

        for _, item := range rubricOrder {
                key, name, handler := item.key, item.name, item.handler
                logf.Tracef("processRubric: checking key '" + key + "'", "\x1b[90m")
                if val, exists := args.Rubric[key]; exists {
                        logf.Tracef("processRubric: key '" + key + "' exists with value " + fmt.Sprintf("%v", val), "\x1b[90m")
                        if boolVal, ok := val.(bool); ok && !boolVal {
                                logf.Tracef("processRubric: key '" + key + "' is false, calling handler '" + name + "'", "\x1b[90m")
                                text, structured := handler()
                                result := map[string]any{
                                        "trunkform": map[string]any{
                                                "version": args.Version, "prompt": args.Prompt,
                                                "steering": args.Steering, "related": args.Related, "rubric": args.Rubric,
                                        },
                                }
                                for k, v := range structured {
                                        result[k] = v
                                }
                                logf.Tracef("processRubric: returning result for '" + name + "'", "\x1b[90m")
                                return mcp.NewToolResultStructured(result, "**"+name+"**: "+text+" In 50 words or less."), nil
                        }
                }
        }
        logf.Tracef("processRubric: all rubric items completed", "\x1b[90m")
        return mcp.NewToolResultText("All rubric items completed. In 50 words or less."), nil
}

func (TrunkformTool) Register(s *server.MCPServer) {
        logf.Tracef("TrunkformTool.Register: registering trunkform tool", "\x1b[90m")
        t := mcp.NewTool(
                "trunkform",
                mcp.WithDescription(`Returns instructions for the AI assistant to act on behalf of user's needs when passed fields from the .trunkform YAML file in the current folder.
                NEVER guess at input values - always read the .trunkform file from disk and use those exact values, and if the .trunkform file does not exist, the
                agent should create the .trunkform file in the current directory with the following schema and default values:
                ---
                version: 0.1.0
                prompt: null
                steering: null
                related: []
                rubric:
                        perf-test-implemented: false
                        unit-test-implemented: false
                        continuous-integration-iac: false
                        continuous-integration-test-double-implemented: false
                        continuous-integration-mocks-implemented: false
                        continuous-integration-iac: false
                        ci-cd-automation-boilerplate: true
                ---
                `),
                mcp.WithString("version", mcp.Description("Version from .trunkform YAML file in the current folder: .version")),
                mcp.WithString("prompt", mcp.Description("Prompt text from .trunkform YAML file in the current folder: .prompt")),
                mcp.WithString("steering", mcp.Description("Steering directive from .trunkform YAML file in the current folder: .steering")),
                mcp.WithArray("related", mcp.Description("Related files array from .trunkform YAML file in the current folder: .related"),
                        mcp.Items(map[string]any{"type": "string"})),
                mcp.WithObject("rubric", mcp.Description("Rubric object from .trunkform YAML file in the current folder: .rubric"),
                        mcp.AdditionalProperties(map[string]any{"type": "boolean"})),
        )

        s.AddTool(t, mcp.NewTypedToolHandler[formatArgs](
                func(ctx context.Context, req mcp.CallToolRequest, args formatArgs) (*mcp.CallToolResult, error) {
                        logf.Tracef("trunkform tool called with rubric: " + fmt.Sprintf("%v", args.Rubric), "\x1b[90m")
                        return processRubric(args)
                },
        ))
        logf.Tracef("TrunkformTool.Register: trunkform tool registered", "\x1b[90m")
